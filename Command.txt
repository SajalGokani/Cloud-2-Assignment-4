# 6. Test direct coupon import via /coupons/import
API_ID=$(cat api_id.txt)
curl -s -X POST "http://localhost:4566/restapis/${API_ID}/prod/_user_request_/coupons/import" \
  -H "Content-Type: application/json" \
  -d '{"coupon_id": "319326de971100158f6fa0daeec978fbalf17e4b", "discount": "25%", "description": "Winter Clearance Coupon"}'

# 7. Request pre-signed URL for coupon.json and upload via HTTP PUT
PRESIGNED_URL=$(curl -s -X POST "http://localhost:4566/restapis/${API_ID}/prod/_user_request_/coupons/import_presigned_url" \
  -H "Content-Type: application/json" \
  -d '{"filename": "coupon.json"}' | python3 -c "import sys, json; print(json.load(sys.stdin)['presigned_url'])")

curl -i -X PUT "$PRESIGNED_URL" \
  -H "Content-Type: application/json" \
  --upload-file coupon.json
